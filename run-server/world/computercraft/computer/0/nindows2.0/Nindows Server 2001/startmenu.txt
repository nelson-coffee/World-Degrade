local screenWidth, screenHeight = term.getSize()
local taskbarHeight = 0
local taskbarBackgroundColor = colors.gray
local taskbarTextColor = colors.black
local startButtonWidth = 14
local startMenuWidth = 25
local startMenuHeight = 15
local startButtonX = 1
local startButtonY = screenHeight - taskbarHeight
local startMenuX = startButtonX
local startMenuY = startButtonY - startMenuHeight - 1

local programs = {
    { name = "App Center", command = "edit" },
    -- Add more program entries here
}

local function clearScreen()
    term.setBackgroundColor(colors.blue)
    term.clear()
    term.setTextColor(colors.white)
end

local function drawTaskbar()
    term.setCursorPos(1, screenHeight - taskbarHeight + 1)
    term.setBackgroundColor(taskbarBackgroundColor)
    term.setTextColor(taskbarTextColor)
    term.clearLine()
    
    
    term.write(taskbarText)
end

local function drawStartButton()
    term.setCursorPos(startButtonX, startButtonY)
    term.setBackgroundColor(colors.gray)
    term.setTextColor(colors.black)
    term.clearLine()
    term.write("Start")
end

local function drawStartMenu()
    term.setCursorPos(startMenuX, startMenuY)
    term.setTextColor(colors.black)

    term.setBackgroundColor(colors.lightGray)
    for y = startMenuY, startMenuY + startMenuHeight - 0 do
        term.setCursorPos(startMenuX, y)
        term.write(string.rep(" ", startMenuWidth))
    end
    
    local startY = startMenuY + 2

    for index, program in ipairs(programs) do
        term.setCursorPos(startMenuX + 1, startY)
        term.write("[" .. program.name .. "]")
        startY = startY + 1
    end
    
    term.setCursorPos(startMenuX + 1, startMenuY + startMenuHeight - 2)
    term.write("")
    term.setCursorPos(startMenuX + 1, startMenuY + startMenuHeight - 1)
    term.write("")
end

while true do
    clearScreen()
    
    if not isStartMenuOpen then
        drawTaskbar()
        drawStartButton()
    else
        drawTaskbar()
        drawStartButton()
        drawStartMenu()
    end

    local event, button, x, y = os.pullEvent("mouse_click")
    
    if button == 1 then -- Check for left-click
        if y == startButtonY and x <= startButtonWidth then
            -- Start button clicked
            isStartMenuOpen = not isStartMenuOpen
        elseif y >= startMenuY + 2 and y <= startMenuY + startMenuHeight - 1
                and x >= startMenuX + 1 and x <= startMenuX + startMenuWidth - 2
                and isStartMenuOpen then
            -- Click on start menu program icons
            local programIndex = y - startMenuY - 1
            local program = programs[programIndex]
            if program then
                shell.run("AppCenter.lua")
            end
            isStartMenuOpen = false
        elseif y == startMenuY + startMenuHeight - 2
                and x >= startMenuX + 1 and x <= startMenuX + startMenuWidth - 2
                and isStartMenuOpen then
            -- Click on shut down button
            os.shutdown()
        elseif y == startMenuY + startMenuHeight - 1
                and x >= startMenuX + 1 and x <= startMenuX + startMenuWidth - 2
                and isStartMenuOpen then
            -- Click on restart button
            os.reboot()
        end
    end
end
